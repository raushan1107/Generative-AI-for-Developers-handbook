Build · Milestone 4: RR Finance Assist Gets Tools

Version 0.4 stops reciting policy and starts answering with facts. RR Finance Assist can look up the signed-in customer's accounts, cards and payments, quote a fee refund or check a dispute with policy code, search the product catalogue and raise a case for the human team, all through strict tools in a loop you control. Triage moves to structured outputs, so the repair loop becomes a fallback.

7 strict toolsStreaming tool loopIdentity from sessionPolicy as codeStructured-output triage≈ 75 min
Plan 10 min · Build 50 min · Verify 15 min
1

Plan

From "I can't see your booking" to "here's your number".

RR Finance Bank· the brief for milestone 4

Pilot feedback is blunt: customers don't want the overdraft policy read to them, they want to know which of their fees come back. The contact-centre lead wants Assist to answer status questions (cards, recent payments) by itself and to open a case, with a clear summary, for anything that needs a person. Compliance insists fee refunds and dispute deadlines come from the same rules the bank's systems use. Security insists that one customer can never see another customer's accounts, whatever they type.

    Policy becomes code. This is the only place fee refunds and dispute windows are decided; fee_refund_quote() is the Drill 03.7 case, now exact.

    """RR Finance Bank policy as code (fictional). Money and dates are decided here, never by the model."""
from datetime import date, timedelta

FEE_USD = 29                  # overdraft fee per overdrawn payment
FREE_BUFFER_USD = 50          # no fee if overdrawn by this much or less at the end of the day
MAX_FEES_PER_DAY = 3
DISPUTE_WINDOW_DAYS = 60      # stand-in rule: the payment date counts as the statement date


def fee_refund_quote(overdraft_days: list[dict], courtesy_used: bool) -> dict:
    """Which overdraft fees were charged in error (always refunded), plus one courtesy refund
    per 12 months if it hasn't been used. The Drill 03.7 case, decided by code."""
    charged = errors = 0
    lines = []
    for d in overdraft_days:
        allowed = 0 if -d["end_of_day_balance_usd"] <= FREE_BUFFER_USD else MAX_FEES_PER_DAY
        wrong = max(0, d["fees_charged"] - allowed)
        charged += d["fees_charged"]
        errors += wrong
        lines.append({"date": d["date"], "fees_charged": d["fees_charged"], "charged_in_error": wrong})
    courtesy = 1 if not courtesy_used and charged > errors else 0
    refund_fees = errors + courtesy
    return {"fees_charged": charged, "fees_charged_usd": charged * FEE_USD,
            "charged_in_error": errors, "courtesy_refund": bool(courtesy),
            "refund_usd": refund_fees * FEE_USD, "days": lines,
            "note": (f"{errors} fee(s) charged in error are refunded" if errors else "no fees were charged in error")
                    + ("; plus one courtesy refund" if courtesy else "")
                    + ("; the courtesy refund was already used this year" if courtesy_used else "")}


def dispute_quote(payment_date: date, today: date) -> dict:
    days = (today - payment_date).days
    if days > DISPUTE_WINDOW_DAYS:
        return {"allowed": False, "days_since_payment": days,
                "note": f"Disputes must be raised within {DISPUTE_WINDOW_DAYS} days; a case can ask the team to review it."}
    return {"allowed": True, "days_since_payment": days,
            "provisional_credit_by": (today + timedelta(days=14)).isoformat(),
            "note": "Inside the 60-day window: a provisional credit within 10 business days."}

    app/toolkit.py is your Drill 04.7 toolkit.py, copied into the app package unchanged. The tools use it:

    """RR Finance Assist tools (milestone 4).
The model can ask for these; our code decides whether and how they run.
Identity rule: the signed-in customer comes from `context`, set by the app, never from the model."""
import json
import re
import time
import uuid
from datetime import date, timedelta
from pathlib import Path
from typing import Literal, Optional

from pydantic import BaseModel, Field, field_validator

from app import backend, policy
from app.config import TICKETS_FILE
from app.toolkit import Toolkit

kit = Toolkit()
ACCOUNT_ID = re.compile(r"AC-\d{4}")
CARD = re.compile(r"\d{4}")
TXN_ID = re.compile(r"TX-\d{4}")

# Friendly status lines the UI shows while a tool runs.
LABELS = {
    "search_products": "Searching the product catalogue",
    "get_product": "Looking up the product",
    "list_my_accounts": "Checking your accounts and cards",
    "list_my_transactions": "Looking at your recent payments",
    "get_card_status": "Checking your card",
    "quote_fee_refund": "Checking your overdraft fees under the policy",
    "quote_dispute": "Checking whether the payment can be disputed",
    "create_support_case": "Creating a case for the support team",
}


def _signed_in(context: dict) -> str:
    customer_id = context.get("customer_id")
    if customer_id not in backend.CUSTOMERS:
        raise PermissionError("no customer is signed in; ask them to sign in first")
    return customer_id


def _owned(table: dict, key: str, context: dict, what: str) -> dict:
    customer_id = _signed_in(context)
    record = table.get(key)
    if record is None or record["customer_id"] != customer_id:   # same answer for both: don't leak which IDs exist
        raise PermissionError(f"{key} is not one of the signed-in customer's {what}")
    return record


def _own_card(last4: str, context: dict) -> dict:
    return _owned(backend.CARDS, last4, context, "cards")


def _own_account(account_id: str, context: dict) -> dict:
    return _owned(backend.ACCOUNTS, account_id, context, "accounts")


def _own_transaction(txn_id: str, context: dict) -> dict:
    t = backend.TRANSACTIONS.get(txn_id)
    if t is None:
        _signed_in(context)
        raise PermissionError(f"{txn_id} is not a payment on the signed-in customer's cards")
    _owned(backend.CARDS, t["card"], context, "cards")
    return t


def _upper_match(pattern, example):
    def check(cls, v):
        v = v.strip().upper()
        if not pattern.fullmatch(v):
            raise ValueError(f"must look like {example}")
        return v
    return classmethod(check)


# ---------------------------------------------------------------- catalogue (no identity needed)
class SearchProductsArgs(BaseModel):
    family: Optional[Literal["accounts", "cards", "loans"]] = None
    audience: Optional[Literal["personal", "student", "business"]] = None
    max_annual_fee_usd: Optional[int] = Field(default=None, description="Highest annual fee in USD, or null")
    text: Optional[str] = Field(default=None, description="One keyword to match in names and descriptions")


@kit.register(SearchProductsArgs)
def search_products(args):
    """Search the RR Finance Bank product catalogue by family, audience, maximum annual fee and/or a keyword."""
    hits = [p for p in backend.PRODUCTS.values()
            if (not args.family or p["family"] == args.family)
            and (not args.audience or p["audience"] == args.audience)
            and (args.max_annual_fee_usd is None or p["annual_fee_usd"] <= args.max_annual_fee_usd)
            and (not args.text or args.text.lower() in (p["name"] + " " + p["description"]).lower())]
    return [{k: p[k] for k in ("id", "name", "family", "audience", "annual_fee_usd", "rate_pct", "rate_type")}
            for p in hits[:8]]


class ProductArgs(BaseModel):
    product_id: str = Field(description="Product ID like P110")


@kit.register(ProductArgs)
def get_product(args):
    """Full details of one product: description, fee, rate and who it is for."""
    return backend.PRODUCTS.get(args.product_id.strip().upper()) or {"error": f"no product {args.product_id}"}


# ---------------------------------------------------------------- the signed-in customer's data
class NoArgs(BaseModel):
    pass


@kit.register(NoArgs, needs_context=True)
def list_my_accounts(args, context):
    """List the signed-in customer's accounts (account_id, product, balance) and cards (card_last4, product, status)."""
    customer_id = _signed_in(context)
    return {"accounts": backend.accounts_of(customer_id), "cards": backend.cards_of(customer_id)}


class CardArg(BaseModel):
    card_last4: str = Field(description="Last four digits of one of the customer's cards, like 4821")
    _check = field_validator("card_last4")(_upper_match(CARD, "4821"))


@kit.register(CardArg, needs_context=True)
def list_my_transactions(args, context):
    """Payments on one of the signed-in customer's cards in the last 90 days: txn_id, date, merchant, amount."""
    _own_card(args.card_last4, context)
    since = (date.today() - timedelta(days=90)).isoformat()
    rows = [{"txn_id": tid, **t} for tid, t in backend.TRANSACTIONS.items()
            if t["card"] == args.card_last4 and t["date"] >= since]
    return sorted(rows, key=lambda r: r["date"], reverse=True)


@kit.register(CardArg, needs_context=True)
def get_card_status(args, context):
    """Status of one of the signed-in customer's cards: active, frozen or replacement sent (with its delivery date)."""
    card = _own_card(args.card_last4, context)
    return {"card_last4": args.card_last4, "product": backend.PRODUCTS[card["product_id"]]["name"],
            "status": card["status"], **({"replacement": card["replacement"]} if "replacement" in card else {})}


class AccountArg(BaseModel):
    account_id: str = Field(description="The customer's account ID, like AC-1234, from list_my_accounts")
    _check = field_validator("account_id")(_upper_match(ACCOUNT_ID, "AC-1234"))


@kit.register(AccountArg, needs_context=True)
def quote_fee_refund(args, context):
    """Overdraft fees on this account: which were charged in error and what RR Finance Bank would refund,
    under the fee policy. Always use this for fee refund amounts; never calculate them yourself."""
    a = _own_account(args.account_id, context)
    quote = policy.fee_refund_quote(backend.OVERDRAFT_DAYS.get(args.account_id, []), a["courtesy_refund_used"])
    return {"account_id": args.account_id, **quote}


class TransactionArg(BaseModel):
    txn_id: str = Field(description="Transaction ID like TX-1002, from list_my_transactions")
    _check = field_validator("txn_id")(_upper_match(TXN_ID, "TX-1234"))


@kit.register(TransactionArg, needs_context=True)
def quote_dispute(args, context):
    """Whether one of the customer's card payments can still be disputed today, and when a provisional credit would arrive."""
    t = _own_transaction(args.txn_id, context)
    return {"txn_id": args.txn_id, "merchant": t["merchant"], "amount_usd": t["amount_usd"],
            **policy.dispute_quote(date.fromisoformat(t["date"]), date.today())}


class CaseArgs(BaseModel):
    category: Literal["card", "dispute", "fraud", "fees", "transfers", "loans", "account", "other"]
    summary: str = Field(description="One or two sentences for the support team, in English")
    reference: Optional[str] = Field(default=None, description="Related account ID, card_last4 or txn_id, or null")
    urgency: Literal["low", "medium", "high"]

    @field_validator("summary")
    @classmethod
    def summary_length(cls, v):
        v = v.strip()
        if not 10 <= len(v) <= 400:
            raise ValueError("summary must be 10 to 400 characters")
        return v


@kit.register(CaseArgs, needs_context=True)
def create_support_case(args, context):
    """Create a case for the human support team. Use it for disputes and fraud you can't settle, fee exceptions,
    anything you can't resolve, or when the customer asks for a person. Tell the customer the case ID."""
    customer_id = _signed_in(context)
    ref = (args.reference or "").strip().upper()
    if ACCOUNT_ID.fullmatch(ref):
        _own_account(ref, context)
    elif CARD.fullmatch(ref):
        _own_card(ref, context)
    elif TXN_ID.fullmatch(ref):
        _own_transaction(ref, context)
    case = {"case_id": "CS-" + uuid.uuid4().hex[:6].upper(), "created": time.strftime("%Y-%m-%dT%H:%M:%S"),
            "customer_id": customer_id, **args.model_dump()}
    path = Path(TICKETS_FILE)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:        # Module 07 adds human approval before actions like this
        f.write(json.dumps(case, ensure_ascii=False) + "\n")
    return {"case_id": case["case_id"], "status": "created", "team": args.category}


TOOLS = kit.schemas()

    The streaming tool loop

    The biggest change. Two details are new: tool calls arrive in pieces while streaming (id and name first, arguments a few characters at a time), so _model_round() assembles them by index; and yield from passes text deltas straight to the browser while the loop keeps the tool calls for itself. Only the final answer goes into history, so memory stays small.

    """Chat service: conversation memory, streaming, error handling and usage logging.
Milestone 3 added a versioned system prompt and an output guard.
Milestone 4 adds tools: the model can look things up and create cases, inside a loop we control."""
import time
import uuid
from types import SimpleNamespace

import openai

FRIENDLY = {
    "rate_limited": "RR Finance Assist is very busy right now. Please try again in a moment.",
    "unreachable": "I couldn't reach the AI service. Please try again.",
    "bad_request": "I couldn't process that message. Could you rephrase it?",

SAFE_REPLY = ("Sorry, I can't share that. Could you rephrase your question, "
              "or shall I raise a case for our support team?")
TOO_MANY_STEPS = "Sorry, that took more steps than I'm allowed. Shall I raise a case for our support team?"


class Turn:
    """What we track during one customer turn, across several model calls."""
    def __init__(self):
        self.id, self.start = uuid.uuid4().hex[:8], time.perf_counter()
        self.ttft, self.parts, self.tools = None, [], []
        self.tokens_in = self.tokens_out = None

    def add_usage(self, u):
        if u:
            self.tokens_in = (self.tokens_in or 0) + u.prompt_tokens
            self.tokens_out = (self.tokens_out or 0) + u.completion_tokens


class ChatService:
    def __init__(self, client=None):
        self.client = client or get_client()
        # session_id -> list of user/assistant messages. In memory for now; Module 08 moves this to a database.
        self.sessions: dict[str, list[dict]] = {}

    def reset(self, session_id: str) -> None:
        self.sessions.pop(session_id, None)

    def _model_round(self, messages, turn):
        """One streamed model call. Yields "delta" events as text arrives and returns
        (tool_calls, finish_reason) when the stream ends."""
        stream = self.client.chat.completions.create(
            model=MODEL, messages=messages, tools=TOOLS, parallel_tool_calls=False,
            stream=True, stream_options={"include_usage": True}, **TOOL_EXTRAS)
        calls, finish = {}, None
        for chunk in stream:
            if chunk.usage:
                turn.add_usage(chunk.usage)
            if not chunk.choices:
                continue
            choice = chunk.choices[0]
            finish = choice.finish_reason or finish
            delta = choice.delta
            if delta.content:
                turn.ttft = turn.ttft or time.perf_counter() - turn.start
                turn.parts.append(delta.content)
                yield {"type": "delta", "text": delta.content}
            # Tool calls also arrive in pieces: the id and name first, then the arguments a few characters at a time.
            for part in delta.tool_calls or []:
                slot = calls.setdefault(part.index, {"id": "", "name": "", "arguments": ""})
                slot["id"] = part.id or slot["id"]
                if part.function:
                    slot["name"] += part.function.name or ""
                    slot["arguments"] += part.function.arguments or ""
        return [calls[i] for i in sorted(calls)], finish

    def stream(self, session_id: str, user_message: str, customer_id: str | None = None):
        """Yield events for one turn: "delta" text, "tool" status lines, an optional "replace"
        if the guard blocks the answer, then one "done" or "error" event."""
        system_prompt, prompt_version = render("assist_system")
        history = self.sessions.setdefault(session_id, [])
        history[:] = history[-(MAX_TURNS_KEPT * 2):]            # keep whole user/assistant pairs
        history.append({"role": "user", "content": user_message})
        # This turn's working messages: history plus any tool calls and results. Only the final answer is kept.
        messages = [{"role": "system", "content": system_prompt}] + history
        context = {"customer_id": customer_id}                     # identity comes from the app, not the model
        turn, finish, error, detail = Turn(), None, None, ""

        try:
            for round_no in range(MAX_TOOL_ROUNDS + 1):
                text_before = len(turn.parts)
                calls, finish = yield from self._model_round(messages, turn)   # streams text to the caller
                if not calls:                                    # no tools wanted: the answer is complete
                    break
                if round_no == MAX_TOOL_ROUNDS:                  # the model keeps asking: stop the loop
                    turn.parts = [TOO_MANY_STEPS]
                    yield {"type": "replace", "text": TOO_MANY_STEPS}
                    log_event(turn=turn.id, level="warning", event="tool_limit", rounds=round_no)
                    break
                # Send back the assistant's tool request, then one "tool" message per call with the result.
                messages.append({"role": "assistant", "content": "".join(turn.parts[text_before:]) or None,
                                 "tool_calls": [{"id": c["id"], "type": "function",
                                                 "function": {"name": c["name"], "arguments": c["arguments"]}}
                                                for c in calls]})
                for c in calls:
                    yield {"type": "tool", "name": c["name"], "label": LABELS.get(c["name"], "Working")}
                    result = kit.dispatch(c["name"], c["arguments"], context)
                    ok = not result.startswith('{"error"')
                    turn.tools.append(c["name"] if ok else c["name"] + ":error")
                    log_event(turn=turn.id, session=session_id[:8], level="info" if ok else "warning",
                              event="tool", tool=c["name"], ok=ok, detail=None if ok else result[:160])
                    messages.append({"role": "tool", "tool_call_id": c["id"], "content": result})
        except openai.RateLimitError as e:
            error, detail = "rate_limited", str(e)
        except openai.APIConnectionError as e:
            error, detail = "unreachable", str(e)
        except openai.BadRequestError as e:
            error, detail = "bad_request", str(e)
        except (openai.AuthenticationError, openai.PermissionDeniedError, openai.NotFoundError) as e:
            error, detail = "config", str(e)
        except openai.APIStatusError as e:
            error, detail = "server", str(e)

        if error:
            history.pop()
            log_event(turn=turn.id, session=session_id[:8], level="error", error=error, detail=detail[:200])
            yield {"type": "error", "message": FRIENDLY[error]}
            return

        reply = "".join(turn.parts)
        flagged = check_output(reply)
        if flagged:
            reply = SAFE_REPLY
            yield {"type": "replace", "text": SAFE_REPLY}
        history.append({"role": "assistant", "content": reply})

        cost = cost_usd(SimpleNamespace(prompt_tokens=turn.tokens_in, completion_tokens=turn.tokens_out)
                        if turn.tokens_in is not None else None)
        event = {
            "type": "done",
            "finish_reason": finish,
            "ttft_s": round(turn.ttft or 0, 2),
            "total_s": round(time.perf_counter() - turn.start, 2),
            "tokens_in": turn.tokens_in,
            "tokens_out": turn.tokens_out,
            "cost_usd": round(cost, 6) if cost is not None else None,
            "prompt_version": prompt_version,
            "tools": turn.tools,                                 # new in milestone 4
        }
        log_event(turn=turn.id, session=session_id[:8], level="warning" if flagged else "info", model=MODEL,
                  flagged=flagged, **{k: v for k, v in event.items() if k != "type"})
        yield event

CUSTOMER_ID = re.compile(r"RR-\d{5}")
# Small models follow the system role more strictly than instructions buried in a long user message.
SYSTEM = "You are a classification service. You reply with one JSON object only: never code, never explanations."


class Triage(BaseModel):
    model_config = ConfigDict(extra="forbid")          # unexpected keys are an error, not a surprise

    # No pattern/max_length here: Azure structured outputs doesn't support them. Validators below do the job.
    customer_name: Optional[str] = Field(description="First name if given, else null")
    customer_id: Optional[str] = Field(description="Like RR-12345, only if it appears in the message, else null")
    product: Optional[str] = Field(description="Product or card name if mentioned, else null")
    issue_category: Literal["card", "dispute", "fraud", "fees", "transfers", "loans", "account", "other"]
    sentiment: Literal["positive", "neutral", "negative"]
    urgency: Literal["low", "medium", "high"]
    summary: str = Field(description="English, at most 20 words")

    @field_validator("customer_id")
    @classmethod
    def customer_id_format(cls, v):
        if v is not None and not CUSTOMER_ID.fullmatch(v):
            raise ValueError("customer_id must look like RR-12345")
        return v

    @field_validator("summary")
    @classmethod
    def summary_length(cls, v):
        if len(v) > 200:
            raise ValueError("summary must be at most 200 characters")
        return v


class TriageError(Exception):
    """The model didn't return a valid triage result."""


def triage(client, message: str, max_attempts: int = 2):
    """Return (Triage, prompt_version, attempts_used)."""
    prompt, version = render("triage", message=neutralise_tags(message))
    problem = ""
    for attempt in range(1, max_attempts + 1):
        try:
            completion = client.chat.completions.parse(
                model=MODEL, response_format=Triage,
                messages=[{"role": "system", "content": SYSTEM}, {"role": "user", "content": prompt}])
        except openai.BadRequestError:
            return _triage_json_repair(client, message)       # provider/model without structured outputs
        except openai.LengthFinishReasonError:
            problem = "the answer was cut off"
            continue
        except openai.ContentFilterFinishReasonError:
            raise TriageError("the content filter blocked the answer") from None
        except ValidationError as e:                          # shape was fine; a business rule failed
            problem = e.errors()[0]["msg"]
            continue
        msg = completion.choices[0].message
        if msg.refusal:
            raise TriageError(f"model refused: {msg.refusal[:100]}")
        return msg.parsed, version, attempt
    raise TriageError(f"no valid result after {max_attempts} attempts: {problem}")


def _triage_json_repair(client, message: str, max_attempts: int = 2):
    """Milestone 3's approach: ask for JSON in the prompt, validate, repair once."""
    prompt, version = render("triage_fallback", message=neutralise_tags(message))
    messages = [{"role": "system", "content": SYSTEM}, {"role": "user", "content": prompt}]
    problem = ""
    for attempt in range(1, max_attempts + 1):
        response = client.chat.completions.create(model=MODEL, messages=messages)
        text = (response.choices[0].message.content or "").strip()
        try:
            return Triage.model_validate_json(text), version, attempt
        except ValidationError as e:
            first = e.errors()[0]
            problem = f"{'.'.join(str(p) for p in first.get('loc', ())) or 'the output'}: {first['msg']}"
            messages += [{"role": "assistant", "content": text},
                         {"role": "user", "content": f"That output is invalid ({problem}). "
                                                     "Return only the corrected JSON object, with no other text."}]
    raise TriageError(f"invalid JSON after {max_attempts} attempts: {problem}")

from types import SimpleNamespace as NS

import httpx
import openai

REQ = httpx.Request("POST", "https://fake.test/openai/v1/chat/completions")


def text(t):
    return {"text": t}


def tool(name, arguments, call_id="call_1"):
    return {"tools": [(call_id, name, arguments)]}


def _chunk(content=None, tool_calls=None, finish=None):
    return NS(usage=None, choices=[NS(delta=NS(content=content, tool_calls=tool_calls), finish_reason=finish)])


class FakeCompletions:
    def __init__(self, rounds, parsed=None):
        self.rounds, self.parsed, self.calls = list(rounds), list(parsed or []), []

    def create(self, **kwargs):
        self.calls.append(kwargs)
        r = self.rounds.pop(0)
        if r.get("raise") == "rate":
            raise openai.RateLimitError("slow down", response=httpx.Response(429, request=REQ), body=None)
        if not kwargs.get("stream"):
            return NS(choices=[NS(message=NS(content=r["text"], refusal=None), finish_reason="stop")])
        return self._stream(r)

    def _stream(self, r):
        yield NS(usage=None, choices=[])                                   # filter-only chunk, like Azure
        if "text" in r:
            for word in r["text"].split(" "):
                yield _chunk(word + " ")
            yield _chunk(finish="stop")
        for index, (call_id, name, arguments) in enumerate(r.get("tools", [])):
            yield _chunk(tool_calls=[NS(index=index, id=call_id, function=NS(name=name, arguments=""))])
            for i in range(0, len(arguments), 7):                          # arguments arrive in fragments
                yield _chunk(tool_calls=[NS(index=index, id=None, function=NS(name=None, arguments=arguments[i:i + 7]))])
        if "tools" in r:
            yield _chunk(finish="tool_calls")
        yield NS(usage=NS(prompt_tokens=100, completion_tokens=10), choices=[])

    def parse(self, **kwargs):
        self.calls.append(kwargs)
        item = self.parsed.pop(0)
        if isinstance(item, Exception):
            raise item
        refusal = item if isinstance(item, str) else None
        return NS(choices=[NS(message=NS(parsed=None if refusal else item, refusal=refusal))])


class FakeClient:
    def __init__(self, rounds=(), parsed=()):
        self.chat = NS(completions=FakeCompletions(rounds, parsed))
