Module 04 · Build milestone

Build · Milestone 4: Traincrest Assist Gets Tools

Version 0.4 stops reciting policy and starts answering with facts. Traincrest Assist can look up the signed-in learner's enrolments and vouchers, quote a refund or reschedule fee calculated by policy code, search the catalogue and raise a ticket for the human team, all through strict tools in a loop you control. Triage moves to structured outputs, so the repair loop becomes a fallback.

1

Plan

From "I can't see your booking" to "here's your number".

TrainCrest· the brief for milestone 4

Pilot feedback is blunt: learners don't want the cancellation policy read to them, they want their refund amount. The support lead wants Assist to answer status questions (enrolments, vouchers) by itself and to open a ticket, with a clear summary, for anything that needs a person. Finance insists refund figures come from the same rules as billing. Security insists that one learner can never see another learner's data, whatever they type.

    Policy becomes code. This is the only place refund money is decided.

    """TrainCrest policy as code (fictional). Money and dates are decided here, never by the model."""
from datetime import date


def refund_quote(start_date: date, today: date, paid_usd: float) -> dict:
    days = (start_date - today).days
    if days < 0:
        return {"days_before_start": days, "refund_percent": 0, "refund_usd": 0.0,
                "note": "The course has already started, so it can't be cancelled for a refund."}
    if days >= 14:
        percent, note = 100, "Full refund: 14 or more days before the start."
    elif days >= 7:
        percent, note = 50, "50% refund: 7 to 13 days before the start."
    else:
        percent, note = 0, "No refund under 7 days, but the learner can move to a later date once, free."
    return {"days_before_start": days, "refund_percent": percent,
            "refund_usd": round(paid_usd * percent / 100, 2), "note": note}


def reschedule_fee(start_date: date, today: date) -> dict:
    days = (start_date - today).days
    if days < 0:
        return {"allowed": False, "fee_usd": None, "note": "The course has started; rescheduling isn't possible."}
    if days >= 7:
        return {"allowed": True, "fee_usd": 0, "note": "Free: 7 or more days before the start."}
    return {"allowed": True, "fee_usd": 50, "note": "Within 7 days: one reschedule for $50."}

    """Traincrest Assist tools (milestone 4).
The model can ask for these; our code decides whether and how they run.
Identity rule: the signed-in learner comes from `context`, set by the app, never from the model."""
import json
import re
import time
import uuid
from datetime import date
from pathlib import Path
from typing import Literal, Optional

from pydantic import BaseModel, Field, field_validator

from app import backend, policy
from app.config import TICKETS_FILE
from app.toolkit import Toolkit

kit = Toolkit()
ENROLMENT_ID = re.compile(r"TC-\d{5}")

# Friendly status lines the UI shows while a tool runs.
LABELS = {
    "search_courses": "Searching the course catalogue",
    "get_course": "Looking up the course",
    "list_my_enrolments": "Checking your enrolments",
    "get_voucher_status": "Checking your exam voucher",
    "quote_refund": "Working out your refund under the policy",
    "quote_reschedule": "Checking the reschedule fee",
    "create_support_ticket": "Creating a support ticket",
}


def _signed_in(context: dict) -> str:
    learner_id = context.get("learner_id")
    if learner_id not in backend.LEARNERS:
        raise PermissionError("no learner is signed in; ask them to sign in first")
    return learner_id


def _own_enrolment(enrolment_id: str, context: dict) -> dict:
    learner_id = _signed_in(context)
    record = backend.ENROLMENTS.get(enrolment_id)
    if record is None or record["learner_id"] != learner_id:       # same answer for both: don't leak which IDs exist
        raise PermissionError(f"{enrolment_id} is not one of the signed-in learner's enrolments")
    return record


class EnrolmentArg(BaseModel):
    enrolment_id: str = Field(description="The learner's enrolment ID, like TC-12345")

    @field_validator("enrolment_id")
    @classmethod
    def looks_like_an_id(cls, v):
        v = v.strip().upper()
        if not ENROLMENT_ID.fullmatch(v):
            raise ValueError("must look like TC-12345")
        return v


# ---------------------------------------------------------------- catalogue (no identity needed)
class SearchCoursesArgs(BaseModel):
    level: Optional[Literal["beginner", "intermediate", "advanced"]] = None
    max_price_usd: Optional[int] = Field(default=None, description="Highest price in USD, or null")
    text: Optional[str] = Field(default=None, description="One keyword to match in titles and descriptions")


@kit.register(SearchCoursesArgs)
def search_courses(args):
    """Search the TrainCrest course catalogue by level, maximum price and/or a keyword."""
    hits = [c for c in backend.COURSES.values()
            if (not args.level or c["level"] == args.level)
            and (args.max_price_usd is None or c["price_usd"] <= args.max_price_usd)
            and (not args.text or args.text.lower() in (c["title"] + " " + c["description"]).lower())]
    return [{k: c[k] for k in ("id", "title", "level", "format", "price_usd")} for c in hits[:8]]


class CourseArgs(BaseModel):
    course_id: str = Field(description="Course ID like C104")


@kit.register(CourseArgs)
def get_course(args):
    """Full details of one course: description, level, format, duration and price."""
    return backend.COURSES.get(args.course_id.strip().upper()) or {"error": f"no course {args.course_id}"}


# ---------------------------------------------------------------- the signed-in learner's data
class NoArgs(BaseModel):
    pass


@kit.register(NoArgs, needs_context=True)
def list_my_enrolments(args, context):
    """List the signed-in learner's enrolments with course, start date, status and attendance."""
    rows = backend.enrolments_of(_signed_in(context))
    return [{k: r[k] for k in ("enrolment_id", "course_title", "start_date", "status", "attendance_pct")} for r in rows]


@kit.register(EnrolmentArg, needs_context=True)
def get_voucher_status(args, context):
    """Exam voucher status for one of the signed-in learner's enrolments."""
    _own_enrolment(args.enrolment_id, context)
    return backend.VOUCHERS.get(args.enrolment_id, {"status": "none", "note": "this course has no exam voucher"})


@kit.register(EnrolmentArg, needs_context=True)
def quote_refund(args, context):
    """Refund the learner would get if they cancelled this enrolment today, under TrainCrest policy.
    Always use this for refund amounts; never calculate them yourself."""
    e = _own_enrolment(args.enrolment_id, context)
    quote = policy.refund_quote(date.fromisoformat(e["start_date"]), date.today(), e["paid_usd"])
    return {"enrolment_id": args.enrolment_id, "paid_usd": e["paid_usd"], **quote}


@kit.register(EnrolmentArg, needs_context=True)
def quote_reschedule(args, context):
    """Whether this enrolment can move to a later date today, and the fee."""
    e = _own_enrolment(args.enrolment_id, context)
    return {"enrolment_id": args.enrolment_id, **policy.reschedule_fee(date.fromisoformat(e["start_date"]), date.today())}


class TicketArgs(BaseModel):
    category: Literal["scheduling", "billing", "exam_voucher", "certificate", "lab_access", "account", "other"]
    summary: str = Field(description="One or two sentences for the support team, in English")
    enrolment_id: Optional[str] = Field(default=None, description="Related enrolment ID, or null")
    urgency: Literal["low", "medium", "high"]

    @field_validator("summary")
    @classmethod
    def summary_length(cls, v):
        v = v.strip()
        if not 10 <= len(v) <= 400:
            raise ValueError("summary must be 10 to 400 characters")
        return v

    @field_validator("enrolment_id")
    @classmethod
    def id_format(cls, v):
        if v is not None and not ENROLMENT_ID.fullmatch(v.strip().upper()):
            raise ValueError("enrolment_id must look like TC-12345")
        return v.strip().upper() if v else v


@kit.register(TicketArgs, needs_context=True)
def create_support_ticket(args, context):
    """Create a ticket for the human support team. Use it for refunds or cancellations, exceptions,
    anything you can't resolve, or when the learner asks for a person. Tell the learner the ticket ID."""
    learner_id = _signed_in(context)
    if args.enrolment_id:
        _own_enrolment(args.enrolment_id, context)
    ticket = {"ticket_id": "T-" + uuid.uuid4().hex[:6].upper(), "created": time.strftime("%Y-%m-%dT%H:%M:%S"),
              "learner_id": learner_id, **args.model_dump()}
    path = Path(TICKETS_FILE)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:        # Module 07 adds human approval before actions like this
        f.write(json.dumps(ticket, ensure_ascii=False) + "\n")
    return {"ticket_id": ticket["ticket_id"], "status": "created", "team": args.category}

  4. Run

    The streaming tool loop

    The biggest change. Two details are new: tool calls arrive in pieces while streaming (id and name first, arguments a few characters at a time), so _model_round() assembles them by index; and yield from passes text deltas straight to the browser while the loop keeps the tool calls for itself. Only the final answer goes into history, so memory stays small.

class Turn:
    """What we track during one learner turn, across several model calls."""
    def __init__(self):
        self.id, self.start = uuid.uuid4().hex[:8], time.perf_counter()
        self.ttft, self.parts, self.tools = None, [], []
        self.tokens_in = self.tokens_out = None

    def add_usage(self, u):
        if u:
            self.tokens_in = (self.tokens_in or 0) + u.prompt_tokens
            self.tokens_out = (self.tokens_out or 0) + u.completion_tokens


class ChatService:
    def __init__(self, client=None):
        self.client = client or get_client()
        # session_id -> list of user/assistant messages. In memory for now; Module 08 moves this to a database.
        self.sessions: dict[str, list[dict]] = {}

    def reset(self, session_id: str) -> None:
        self.sessions.pop(session_id, None)

    def _model_round(self, messages, turn):
        """One streamed model call. Yields "delta" events as text arrives and returns
        (tool_calls, finish_reason) when the stream ends."""
        stream = self.client.chat.completions.create(
            model=MODEL, messages=messages, tools=TOOLS, parallel_tool_calls=False,
            stream=True, stream_options={"include_usage": True}, **TOOL_EXTRAS)
        calls, finish = {}, None
        for chunk in stream:
            if chunk.usage:
                turn.add_usage(chunk.usage)
            if not chunk.choices:
                continue
            choice = chunk.choices[0]
            finish = choice.finish_reason or finish
            delta = choice.delta
            if delta.content:
                turn.ttft = turn.ttft or time.perf_counter() - turn.start
                turn.parts.append(delta.content)
                yield {"type": "delta", "text": delta.content}
            # Tool calls also arrive in pieces: the id and name first, then the arguments a few characters at a time.
            for part in delta.tool_calls or []:
                slot = calls.setdefault(part.index, {"id": "", "name": "", "arguments": ""})
                slot["id"] = part.id or slot["id"]
                if part.function:
                    slot["name"] += part.function.name or ""
                    slot["arguments"] += part.function.arguments or ""
        return [calls[i] for i in sorted(calls)], finish

    def stream(self, session_id: str, user_message: str, learner_id: str | None = None):
        """Yield events for one turn: "delta" text, "tool" status lines, an optional "replace"
        if the guard blocks the answer, then one "done" or "error" event."""
        system_prompt, prompt_version = render("assist_system")
        history = self.sessions.setdefault(session_id, [])
        history[:] = history[-(MAX_TURNS_KEPT * 2):]            # keep whole user/assistant pairs
        history.append({"role": "user", "content": user_message})
        # This turn's working messages: history plus any tool calls and results. Only the final answer is kept.
        messages = [{"role": "system", "content": system_prompt}] + history
        context = {"learner_id": learner_id}                     # identity comes from the app, not the model
        turn, finish, error, detail = Turn(), None, None, ""

        try:
            for round_no in range(MAX_TOOL_ROUNDS + 1):
                text_before = len(turn.parts)
                calls, finish = yield from self._model_round(messages, turn)   # streams text to the caller
                if not calls:                                    # no tools wanted: the answer is complete
                    break
                if round_no == MAX_TOOL_ROUNDS:                  # the model keeps asking: stop the loop
                    turn.parts = [TOO_MANY_STEPS]
                    yield {"type": "replace", "text": TOO_MANY_STEPS}
                    log_event(turn=turn.id, level="warning", event="tool_limit", rounds=round_no)
                    break
                # Send back the assistant's tool request, then one "tool" message per call with the result.
                messages.append({"role": "assistant", "content": "".join(turn.parts[text_before:]) or None,
                                 "tool_calls": [{"id": c["id"], "type": "function",
                                                 "function": {"name": c["name"], "arguments": c["arguments"]}}
                                                for c in calls]})
                for c in calls:
                    yield {"type": "tool", "name": c["name"], "label": LABELS.get(c["name"], "Working")}
                    result = kit.dispatch(c["name"], c["arguments"], context)
                    ok = not result.startswith('{"error"')
                    turn.tools.append(c["name"] if ok else c["name"] + ":error")
                    log_event(turn=turn.id, session=session_id[:8], level="info" if ok else "warning",
                              event="tool", tool=c["name"], ok=ok, detail=None if ok else result[:160])
                    messages.append({"role": "tool", "tool_call_id": c["id"], "content": result})
        except openai.RateLimitError as e:
            error, detail = "rate_limited", str(e)
        except openai.APIConnectionError as e:
            error, detail = "unreachable", str(e)
        except openai.BadRequestError as e:
            error, detail = "bad_request", str(e)
        except (openai.AuthenticationError, openai.PermissionDeniedError, openai.NotFoundError) as e:
            error, detail = "config", str(e)
        except openai.APIStatusError as e:
            error, detail = "server", str(e)

        if error:
            history.pop()
            log_event(turn=turn.id, session=session_id[:8], level="error", error=error, detail=detail[:200])
            yield {"type": "error", "message": FRIENDLY[error]}
            return

        reply = "".join(turn.parts)
        flagged = check_output(reply)
        if flagged:
            reply = SAFE_REPLY
            yield {"type": "replace", "text": SAFE_REPLY}
        history.append({"role": "assistant", "content": reply})

        cost = cost_usd(SimpleNamespace(prompt_tokens=turn.tokens_in, completion_tokens=turn.tokens_out)
                        if turn.tokens_in is not None else None)
        event = {
            "type": "done",
            "finish_reason": finish,
            "ttft_s": round(turn.ttft or 0, 2),
            "total_s": round(time.perf_counter() - turn.start, 2),
            "tokens_in": turn.tokens_in,
            "tokens_out": turn.tokens_out,
            "cost_usd": round(cost, 6) if cost is not None else None,
            "prompt_version": prompt_version,
            "tools": turn.tools,                                 # new in milestone 4
        }
        log_event(turn=turn.id, session=session_id[:8], level="warning" if flagged else "info", model=MODEL,
                  flagged=flagged, **{k: v for k, v in event.items() if k != "type"})
        yield event

import re

from pydantic import BaseModel, ConfigDict, Field, ValidationError, field_validator

from app.config import MODEL

ENROLMENT_ID = re.compile(r"TC-\d{5}")


class Triage(BaseModel):
    model_config = ConfigDict(extra="forbid")          # unexpected keys are an error, not a surprise

    # No pattern/max_length here: Azure structured outputs doesn't support them. Validators below do the job.
    learner_name: Optional[str] = Field(description="First name if given, else null")
    enrolment_id: Optional[str] = Field(description="Like TC-12345, only if it appears in the message, else null")
    course: Optional[str] = Field(description="Course name if mentioned, else null")
    issue_category: Literal["scheduling", "billing", "exam_voucher", "certificate", "lab_access", "account", "other"]
    sentiment: Literal["positive", "neutral", "negative"]
    urgency: Literal["low", "medium", "high"]
    summary: str = Field(description="English, at most 20 words")

    @field_validator("enrolment_id")
    @classmethod
    def enrolment_id_format(cls, v):
        if v is not None and not ENROLMENT_ID.fullmatch(v):
            raise ValueError("enrolment_id must look like TC-12345")
        return v

    @field_validator("summary")
    @classmethod
    def summary_length(cls, v):
        if len(v) > 200:
            raise ValueError("summary must be at most 200 characters")
        return v


class TriageError(Exception):
    """The model didn't return a valid triage result."""


def triage(client, message: str, max_attempts: int = 2):
    """Return (Triage, prompt_version, attempts_used)."""
    prompt, version = render("triage", message=neutralise_tags(message))
    problem = ""
    for attempt in range(1, max_attempts + 1):
        try:
            completion = client.chat.completions.parse(
                model=MODEL, messages=[{"role": "user", "content": prompt}], response_format=Triage)
        except openai.BadRequestError:
            return _triage_json_repair(client, message)       # provider/model without structured outputs
        except openai.LengthFinishReasonError:
            problem = "the answer was cut off"
            continue
        except openai.ContentFilterFinishReasonError:
            raise TriageError("the content filter blocked the answer") from None
        except ValidationError as e:                          # shape was fine; a business rule failed
            problem = e.errors()[0]["msg"]
            continue
        msg = completion.choices[0].message
        if msg.refusal:
            raise TriageError(f"model refused: {msg.refusal[:100]}")
        return msg.parsed, version, attempt
    raise TriageError(f"no valid result after {max_attempts} attempts: {problem}")


def _triage_json_repair(client, message: str, max_attempts: int = 2):
    """Milestone 3's approach: ask for JSON in the prompt, validate, repair once."""
    prompt, version = render("triage_fallback", message=neutralise_tags(message))
    messages = [{"role": "user", "content": prompt}]
    problem = ""
    for attempt in range(1, max_attempts + 1):
        response = client.chat.completions.create(model=MODEL, messages=messages)
        text = (response.choices[0].message.content or "").strip()
        try:
            return Triage.model_validate_json(text), version, attempt
        except ValidationError as e:
            first = e.errors()[0]
            problem = f"{'.'.join(str(p) for p in first.get('loc', ())) or 'the output'}: {first['msg']}"
            messages += [{"role": "assistant", "content": text},
                         {"role": "user", "content": f"That output is invalid ({problem}). "
                                                     "Return only the corrected JSON object, with no other text."}]
    raise TriageError(f"invalid JSON after {max_attempts} attempts: {problem}")

    """A scripted stand-in for the OpenAI client, so tests run offline in milliseconds.
Each create() call plays the next "round": either text or tool calls, streamed in small pieces

from types import SimpleNamespace as NS

import httpx
import openai

REQ = httpx.Request("POST", "https://fake.test/openai/v1/chat/completions")


def text(t):
    return {"text": t}


def tool(name, arguments, call_id="call_1"):
    return {"tools": [(call_id, name, arguments)]}


def _chunk(content=None, tool_calls=None, finish=None):
    return NS(usage=None, choices=[NS(delta=NS(content=content, tool_calls=tool_calls), finish_reason=finish)])


class FakeCompletions:
    def __init__(self, rounds, parsed=None):
        self.rounds, self.parsed, self.calls = list(rounds), list(parsed or []), []

    def create(self, **kwargs):
        self.calls.append(kwargs)
        r = self.rounds.pop(0)
        if r.get("raise") == "rate":
            raise openai.RateLimitError("slow down", response=httpx.Response(429, request=REQ), body=None)
        if not kwargs.get("stream"):
            return NS(choices=[NS(message=NS(content=r["text"], refusal=None), finish_reason="stop")])
        return self._stream(r)

    def _stream(self, r):
        yield NS(usage=None, choices=[])                                   # filter-only chunk, like Azure
        if "text" in r:
            for word in r["text"].split(" "):
                yield _chunk(word + " ")
            yield _chunk(finish="stop")
        for index, (call_id, name, arguments) in enumerate(r.get("tools", [])):
            yield _chunk(tool_calls=[NS(index=index, id=call_id, function=NS(name=name, arguments=""))])
            for i in range(0, len(arguments), 7):                          # arguments arrive in fragments
                yield _chunk(tool_calls=[NS(index=index, id=None, function=NS(name=None, arguments=arguments[i:i + 7]))])
        if "tools" in r:
            yield _chunk(finish="tool_calls")
        yield NS(usage=NS(prompt_tokens=100, completion_tokens=10), choices=[])

    def parse(self, **kwargs):
        self.calls.append(kwargs)
        item = self.parsed.pop(0)
        if isinstance(item, Exception):
            raise item
        refusal = item if isinstance(item, str) else None
        return NS(choices=[NS(message=NS(parsed=None if refusal else item, refusal=refusal))])


class FakeClient:
    def __init__(self, rounds=(), parsed=()):
        self.chat = NS(completions=FakeCompletions(rounds, parsed))
