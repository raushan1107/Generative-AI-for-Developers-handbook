Build · Milestone 3: Prompts as Code, Triage as Data

Version 0.3 makes Traincrest Assist's behaviour deliberate. The system prompt becomes a structured, versioned spec with TrainCrest's policy in it; a new /api/triage endpoint turns any learner message into validated JSON the team can route; a guard checks every reply before learners see it; and a check script tells you, with numbers, whether a prompt change is safe to ship.

Plan

From one vague string to versioned, tested behaviour.

TrainCrest· the brief for milestone 3

The pilot showed three problems. Assist still guesses policy answers. The support team wants every incoming message triaged into structured data (category, urgency, enrolment ID, summary) so it can be routed automatically. And the security lead wants proof that a manipulated reply can't reach learners. Nobody may change a prompt without evidence it still works.

    The guard (from Drill 03.13, plus input neutralising)

    """Guardrails around untrusted text (milestone 3). Module 10 adds more layers."""
import re

URL = re.compile(r"(?:https?://|www\.)\S+|\b(?:[a-z0-9-]+\.)+(?:com|net|org|io|co|xyz|info|example)\b", re.I)
EMAIL = re.compile(r"\b[\w.+-]+@(?:[\w-]+\.)+[\w-]+\b")
PHONE = re.compile(r"(?<!\w)\+?\d(?:[\s-]?\d){8,}(?!\w)")
INJECTION_ECHO = re.compile(r"ignore (?:all |your )?(?:previous |prior )?instructions", re.I)
OUR_TAGS = re.compile(r"</?\s*(?:message|learner_message|policy|ticket)\s*>", re.I)

ALLOWED_DOMAINS = ("traincrest.example",)


def neutralise_tags(text: str) -> str:
    """Remove our own delimiter tags from untrusted text, so it can't 'close' a <message> block
    early and smuggle text in as if it were instructions."""
    return OUR_TAGS.sub("", text)


def _host(url_text: str) -> str:
    host = re.sub(r"^(?:https?://)?(?:www\.)?", "", url_text.lower())
    return re.split(r"[/?#:]", host, maxsplit=1)[0].rstrip(".,;)")


def check_output(text: str) -> list[str]:
    """Return problem names found in model output (empty list = passed)."""
    problems = set()
    if EMAIL.search(text):
        problems.add("email")
    for match in URL.finditer(EMAIL.sub(" ", text)):
        host = _host(match.group(0))
        if not any(host == d or host.endswith("." + d) for d in ALLOWED_DOMAINS):
            problems.add("url")
    if PHONE.search(text):
        problems.add("phone")
    if INJECTION_ECHO.search(text):
        problems.add("injection_echo")
    return sorted(problems)
  4. Run

    Triage: prompt → JSON → Pydantic → repair once

    This is the Drill 03.11 exercise turned into production code. Pydantic replaces the hand-written checker, and one repair attempt shows the model its own invalid output plus the exact problem.

    """Ticket triage (milestone 3): the model returns JSON, Pydantic validates it,
and we ask the model to repair its answer once if validation fails."""
from typing import Literal, Optional

from pydantic import BaseModel, ConfigDict, Field, ValidationError

from app.config import MODEL
from app.guard import neutralise_tags
from app.prompts import render


class Triage(BaseModel):
    model_config = ConfigDict(extra="forbid")          # unexpected keys are an error, not a surprise

    learner_name: Optional[str] = None
    enrolment_id: Optional[str] = Field(default=None, pattern=r"^TC-\d{5}$")
    course: Optional[str] = None
    issue_category: Literal["scheduling", "billing", "exam_voucher", "certificate", "lab_access", "account", "other"]
    sentiment: Literal["positive", "neutral", "negative"]
    urgency: Literal["low", "medium", "high"]
    summary: str = Field(max_length=200)


class TriageError(Exception):
    """The model didn't return valid triage JSON, even after a repair attempt."""


def triage(client, message: str, max_attempts: int = 2):
    """Return (Triage, prompt_version, attempts_used)."""
    prompt, version = render("triage", message=neutralise_tags(message))
    messages = [{"role": "user", "content": prompt}]
    problem = ""
    for attempt in range(1, max_attempts + 1):
        response = client.chat.completions.create(model=MODEL, messages=messages)
        text = (response.choices[0].message.content or "").strip()
        try:
            return Triage.model_validate_json(text), version, attempt
        except ValidationError as e:
            first = e.errors()[0]
            where = ".".join(str(p) for p in first.get("loc", ())) or "the output"
            problem = f"{where}: {first['msg']}"
            # Show the model its own answer and the exact problem, and ask for a corrected version.
            messages += [
                {"role": "assistant", "content": text},
                {"role": "user", "content": f"That output is invalid ({problem}). "
                                            "Return only the corrected JSON object, with no other text."},
            ]
    raise TriageError(f"invalid JSON after {max_attempts} attempts: {problem}")

    Chat uses the library and the guard

    Three changes from Milestone 2: the system prompt comes from render("assist_system"), the finished reply goes through check_output() (a flagged reply is replaced with a safe message via a replace event, which your Milestone 2 UI already handles), and the prompt version is logged.

    """Chat service: conversation memory, streaming, error handling and usage logging.
Milestone 3 adds: a versioned, structured system prompt and an output guard."""
import time
import uuid

import openai

from app.config import MAX_TURNS_KEPT, MODEL
from app.guard import check_output
from app.llm import get_client
from app.logging_utils import cost_usd, log_event
from app.prompts import render

FRIENDLY = {
    "rate_limited": "Traincrest Assist is very busy right now. Please try again in a moment.",
    "unreachable": "I couldn't reach the AI service. Please try again.",
    "bad_request": "I couldn't process that message. Could you rephrase it?",
    "config": "Traincrest Assist is temporarily unavailable. Our team has been notified.",
    "server": "Something went wrong on our side. Please try again shortly.",
}
SAFE_REPLY = ("Sorry, I can't share that. Could you rephrase your question, "
              "or shall I raise a ticket for our support team?")


class ChatService:
    def __init__(self, client=None):
        self.client = client or get_client()
        # session_id -> list of messages. In memory for now; Module 08 moves this to a database.
        self.sessions: dict[str, list[dict]] = {}

    def reset(self, session_id: str) -> None:
        self.sessions.pop(session_id, None)

    def stream(self, session_id: str, user_message: str):
        """Yield events for one turn: {"type": "delta"} while streaming, an optional
        {"type": "replace"} if the guard blocks the answer, then one "done" or "error" event."""
        system_prompt, prompt_version = render("assist_system")
        history = self.sessions.setdefault(session_id, [])
        history[:] = history[-(MAX_TURNS_KEPT * 2):]            # keep whole user/assistant pairs
        history.append({"role": "user", "content": user_message})
        turn_id, start = uuid.uuid4().hex[:8], time.perf_counter()
        ttft, parts, usage, finish = None, [], None, None
        error, detail = None, ""

        try:
            stream = self.client.chat.completions.create(
                model=MODEL,
                messages=[{"role": "system", "content": system_prompt}] + history,
                stream=True,
                stream_options={"include_usage": True},
            )
            for chunk in stream:
                if chunk.usage:
                    usage = chunk.usage
                if not chunk.choices:
                    continue
                choice = chunk.choices[0]
                finish = choice.finish_reason or finish
                if choice.delta.content:
                    ttft = ttft or time.perf_counter() - start
                    parts.append(choice.delta.content)
                    yield {"type": "delta", "text": choice.delta.content}
        except openai.RateLimitError as e:
            error, detail = "rate_limited", str(e)
        except openai.APIConnectionError as e:
            error, detail = "unreachable", str(e)
        except openai.BadRequestError as e:
            error, detail = "bad_request", str(e)
        except (openai.AuthenticationError, openai.PermissionDeniedError, openai.NotFoundError) as e:
            error, detail = "config", str(e)
        except openai.APIStatusError as e:
            error, detail = "server", str(e)

        if error:
            history.pop()
            log_event(turn=turn_id, session=session_id[:8], level="error", error=error, detail=detail[:200])
            yield {"type": "error", "message": FRIENDLY[error]}
            return

        reply = "".join(parts)
        flagged = check_output(reply)
        if flagged:                                              # new in milestone 3
            reply = SAFE_REPLY
            yield {"type": "replace", "text": SAFE_REPLY}
        history.append({"role": "assistant", "content": reply})

        cost = cost_usd(usage)
        event = {
            "type": "done",
            "finish_reason": finish,
            "ttft_s": round(ttft or 0, 2),
            "total_s": round(time.perf_counter() - start, 2),
            "tokens_in": getattr(usage, "prompt_tokens", None),
            "tokens_out": getattr(usage, "completion_tokens", None),
            "cost_usd": round(cost, 6) if cost is not None else None,
            "prompt_version": prompt_version,
        }
        log_event(turn=turn_id, session=session_id[:8], level="warning" if flagged else "info", model=MODEL,
                  flagged=flagged, **{k: v for k, v in event.items() if k != "type"})
        yield event

    """Traincrest Assist API, milestone 3.

Then open http://127.0.0.1:8000 (chat UI) or http://127.0.0.1:8000/docs (try /api/triage there).

import json
from pathlib import Path

import openai
from fastapi import FastAPI, HTTPException
from fastapi.responses import FileResponse, StreamingResponse
from pydantic import BaseModel, Field

from app.chat import ChatService
from app.logging_utils import log_event
from app.triage import TriageError, triage

app = FastAPI(title="Traincrest Assist", version="0.3")
chat = ChatService()
WEB = Path(__file__).resolve().parent.parent / "web"


class ChatRequest(BaseModel):
    session_id: str = Field(min_length=8, max_length=64)
    message: str = Field(min_length=1, max_length=4000)


class ResetRequest(BaseModel):
    session_id: str = Field(min_length=8, max_length=64)


class TriageRequest(BaseModel):
    message: str = Field(min_length=1, max_length=4000)


@app.get("/")
def home():
    return FileResponse(WEB / "index.html")


@app.get("/api/health")
def health():
    return {"status": "ok"}


@app.post("/api/chat")
def chat_endpoint(req: ChatRequest):
    """Stream the answer as server-sent events: one `data: {json}` line per event."""
    def events():
        for event in chat.stream(req.session_id, req.message):
            yield f"data: {json.dumps(event, ensure_ascii=False)}\n\n"
    return StreamingResponse(events(), media_type="text/event-stream",
                             headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})


@app.post("/api/reset")
def reset(req: ResetRequest):
    chat.reset(req.session_id)
    return {"status": "reset"}


@app.post("/api/triage")
def triage_endpoint(req: TriageRequest):
    """New in milestone 3: classify a learner message into validated JSON for routing."""
    try:
        result, version, attempts = triage(chat.client, req.message)
    except TriageError as e:
        log_event(level="warning", event="triage_failed", detail=str(e)[:200])
        raise HTTPException(status_code=502, detail="Could not triage this message. Please try again.")
    except openai.APIError as e:
        log_event(level="error", event="triage_api_error", error=type(e).__name__)
        raise HTTPException(status_code=503, detail="The AI service is unavailable. Please try again shortly.")
    log_event(level="info", event="triage", prompt_version=version, attempts=attempts,
              category=result.issue_category, urgency=result.urgency)
    return {"result": result.model_dump(), "prompt_version": version, "attempts": attempts}

    """Reliability check for triage: run it before every prompt or model change.

From the project folder:   python -m tests.check_triage            (calls the real model)
                           python -m tests.check_triage --min-accuracy 0.9
Exit code 0 = pass, 1 = fail, so a CI pipeline can block a bad change (Module 12).
"""
import argparse
import json
import sys
from pathlib import Path

from app.llm import get_client
from app.triage import TriageError, triage


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--data", default="data/eval_tickets.json")
    parser.add_argument("--min-accuracy", type=float, default=0.85)
    args = parser.parse_args()

    rows = json.loads(Path(args.data).read_text(encoding="utf-8"))
    client = get_client()
    valid = correct = repaired = 0
    version = "?"
    for row in rows:
        try:
            result, version, attempts = triage(client, row["text"])
        except TriageError as e:
            print(f"  INVALID {row['text'][:50]}  ({e})")
            continue
        valid += 1
        repaired += attempts > 1
        ok = result.issue_category == row["label"]
        correct += ok
        print(f"  {'✓' if ok else '✗'} {result.issue_category:13} expected {row['label']:13} {row['text'][:44]}")

    n = len(rows)
    accuracy = correct / n
    print(f"\nprompt {version}: valid {valid}/{n} · repaired {repaired} · accuracy {correct}/{n} ({accuracy:.0%})")
    if valid < n or accuracy < args.min_accuracy:
        print(f"FAIL (needs 100% valid and at least {args.min_accuracy:.0%} accuracy)")
        return 1
    print("PASS")
    return 0
