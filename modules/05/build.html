import re
from pathlib import Path

import numpy as np
from rank_bm25 import BM25Okapi

from app import backend

HELP_DIR = Path(__file__).resolve().parent.parent / "data" / "help"
STOP = set("a an and are as at be by can do does for from how i if in is it me my of on or our the there "
           "this to use we what when where which will with you your".split())


# ---------------------------------------------------------------- documents
def chunk_by_headings(markdown: str):
    """One chunk per '## ' section, prefixed with 'Article > Section' so it stands on its own.
    Drill 05.7 showed this beat fixed-size chunks on our help centre."""
    title = re.search(r"^# (.+)$", markdown, re.M).group(1).strip()
    for part in re.split(r"^## ", markdown, flags=re.M)[1:]:
        heading, _, body = part.partition("\n")
        yield heading.strip(), f"{title} > {heading.strip()}\n{body.strip()}"


def load_documents():
    """Everything we search, as plain dicts: one per product and one per help-article section."""
    docs = []
    for p in backend.PRODUCTS.values():
        docs.append({"id": p["id"], "kind": "product", "title": p["name"],
                     "text": f"{p['name']}. {p['description']}",
                     "family": p["family"], "audience": p["audience"], "annual_fee_usd": p["annual_fee_usd"]})
    for path in sorted(HELP_DIR.glob("*.md")):
        for n, (heading, text) in enumerate(chunk_by_headings(path.read_text(encoding="utf-8")), 1):
            docs.append({"id": f"{path.stem}-{n}", "kind": "help", "title": heading, "text": text,
                         "source": f"{path.stem}.md", "family": None, "audience": None, "annual_fee_usd": None})
    return docs


def embed(client, texts, batch=64) -> np.ndarray:
    vectors = []
    for i in range(0, len(texts), batch):
        response = client.embeddings.create(model=EMBED_MODEL, input=texts[i:i + batch])
        vectors += [d.embedding for d in response.data]
    v = np.array(vectors, dtype=np.float32)
    return v / np.linalg.norm(v, axis=1, keepdims=True)


def tokenize(text: str):
    words = re.findall(r"[a-z0-9]+(?:-[a-z0-9]+)*", text.lower())
    return [w[:-1] if len(w) > 3 and w.endswith("s") and not w.endswith("ss") else w
            for w in words if w not in STOP]


def rrf(rankings, k=60):
    totals = {}
    for ranking in rankings:
        for rank, i in enumerate(ranking, start=1):
            totals[i] = totals.get(i, 0.0) + 1.0 / (k + rank)
    return sorted(totals.items(), key=lambda kv: -kv[1])


def _matches(doc, filters):
    if filters.get("kind") and doc["kind"] != filters["kind"]:
        return False
    if filters.get("family") and doc["family"] != filters["family"]:
        return False
    if filters.get("audience") and doc["audience"] != filters["audience"]:
        return False
    if filters.get("max_annual_fee_usd") is not None and (doc["annual_fee_usd"] is None or doc["annual_fee_usd"] > filters["max_annual_fee_usd"]):
        return False
    return True


# ---------------------------------------------------------------- local backend
class LocalIndex:
    def __init__(self, docs, vectors, model):
        self.docs, self.vectors, self.model = docs, vectors, model
        self.bm25 = BM25Okapi([tokenize(d["text"]) for d in docs])

    @classmethod
    def build(cls, client):
        docs = load_documents()
        return cls(docs, embed(client, [d["text"] for d in docs]), EMBED_MODEL)

    def save(self, folder=INDEX_DIR):
        folder = Path(folder)
        folder.mkdir(parents=True, exist_ok=True)
        np.save(folder / "vectors.npy", self.vectors)
        (folder / "docs.json").write_text(json.dumps({"model": self.model, "docs": self.docs}, ensure_ascii=False),
                                          encoding="utf-8")

    @classmethod
    def load(cls, folder=INDEX_DIR):
        folder = Path(folder)
        saved = json.loads((folder / "docs.json").read_text(encoding="utf-8"))
        if saved["model"] != EMBED_MODEL:      # vectors from different models can't be compared
            raise RuntimeError(f"index was built with {saved['model']}, but EMBED_MODEL is {EMBED_MODEL}; rebuild it")
        return cls(saved["docs"], np.load(folder / "vectors.npy"), saved["model"])

    def search(self, query, query_vector, k=5, **filters):
        allowed = [i for i, d in enumerate(self.docs) if _matches(d, filters)]     # pre-filter
        if not allowed:
            return []
        cosine = self.vectors[allowed] @ query_vector
        vector_rank = [allowed[j] for j in np.argsort(-cosine)]
        keyword_scores = self.bm25.get_scores(tokenize(query))
        keyword_rank = sorted((i for i in allowed if keyword_scores[i] > 0), key=lambda i: -keyword_scores[i])
        cos_of = dict(zip(allowed, cosine))
        return [{**self.docs[i], "score": round(score, 4), "cosine": round(float(cos_of[i]), 3)}
                for i, score in rrf([keyword_rank, vector_rank])[:k]]


# ---------------------------------------------------------------- Azure AI Search backend
class AzureIndex:
    """Same search() as LocalIndex, done by Azure AI Search (hybrid = keyword + vector, merged with RRF)."""
    def __init__(self):
        from azure.core.credentials import AzureKeyCredential
        from azure.search.documents import SearchClient
        self.client = SearchClient(AZURE_SEARCH_ENDPOINT, AZURE_SEARCH_INDEX, AzureKeyCredential(AZURE_SEARCH_API_KEY))

    def search(self, query, query_vector, k=5, **filters):
        from azure.search.documents.models import VectorizedQuery
        clauses = []
        for field in ("kind", "family", "audience"):
            if filters.get(field):
                clauses.append(f"{field} eq '{filters[field]}'")          # values come from enums, not free text
        if filters.get("max_annual_fee_usd") is not None:
            clauses.append(f"annual_fee_usd le {int(filters['max_annual_fee_usd'])}")
        results = self.client.search(
            search_text=query, top=k, filter=" and ".join(clauses) or None,
            vector_queries=[VectorizedQuery(vector=query_vector.tolist(), k_nearest_neighbors=k, fields="contentVector")])
        return [{**{key: r.get(key) for key in ("id", "kind", "title", "text", "source", "family", "audience", "annual_fee_usd")},
                 "score": round(r["@search.score"], 4)} for r in results]


# ---------------------------------------------------------------- the service the app uses
class SearchService:
    def __init__(self, client, index=None):
        self.client = client
        if index is not None:
            self.index = index
        elif SEARCH_BACKEND == "azure":
            self.index = AzureIndex()
        else:
            self.index = LocalIndex.load()

    def search(self, query: str, k: int = 5, **filters):
        query_vector = embed(self.client, [query])[0]
        return self.index.search(query, query_vector, k=k, **filters)

    """Build the search index for RR Finance Assist (milestone 5).

From the project folder:  python -m scripts.build_index
SEARCH_BACKEND=local  -> embeds products and help sections, saves data/index/ (vectors.npy + docs.json)
SEARCH_BACKEND=azure  -> creates or updates the Azure AI Search index and uploads the same documents
Re-run it whenever products or help articles change, or when you switch embedding model."""
import time

from app.config import AZURE_SEARCH_API_KEY, AZURE_SEARCH_ENDPOINT, AZURE_SEARCH_INDEX, EMBED_MODEL, SEARCH_BACKEND
from app.llm import get_client
from app.search import LocalIndex, embed, load_documents


def build_azure(client):
    from azure.core.credentials import AzureKeyCredential
    from azure.search.documents import SearchClient
    from azure.search.documents.indexes import SearchIndexClient
    from azure.search.documents.indexes.models import (HnswAlgorithmConfiguration, SearchableField, SearchField,
                                                       SearchFieldDataType, SearchIndex, SimpleField, VectorSearch,
                                                       VectorSearchProfile)
    docs = load_documents()
    vectors = embed(client, [d["text"] for d in docs])
    s = SearchFieldDataType
    fields = [
        SimpleField(name="id", type=s.String, key=True),
        SimpleField(name="kind", type=s.String, filterable=True),
        SearchableField(name="title", type=s.String),
        SearchableField(name="text", type=s.String, analyzer_name="en.microsoft"),   # stemming, stop words
        SimpleField(name="source", type=s.String),
        SimpleField(name="family", type=s.String, filterable=True),
        SimpleField(name="audience", type=s.String, filterable=True),
        SimpleField(name="annual_fee_usd", type=s.Int32, filterable=True, sortable=True),
        SearchField(name="contentVector", type=s.Collection(s.Single), searchable=True,
                    vector_search_dimensions=vectors.shape[1], vector_search_profile_name="vprofile"),
    ]
    index = SearchIndex(name=AZURE_SEARCH_INDEX, fields=fields, vector_search=VectorSearch(
        algorithms=[HnswAlgorithmConfiguration(name="hnsw")],
        profiles=[VectorSearchProfile(name="vprofile", algorithm_configuration_name="hnsw")]))
    credential = AzureKeyCredential(AZURE_SEARCH_API_KEY)
    SearchIndexClient(AZURE_SEARCH_ENDPOINT, credential).create_or_update_index(index)
    upload = [{**d, "contentVector": v.tolist()} for d, v in zip(docs, vectors)]
    results = SearchClient(AZURE_SEARCH_ENDPOINT, AZURE_SEARCH_INDEX, credential).merge_or_upload_documents(upload)
    print(f"Azure AI Search index '{AZURE_SEARCH_INDEX}': {sum(r.succeeded for r in results)}/{len(upload)} documents.")


def main():
    client, start = get_client(), time.perf_counter()
    if SEARCH_BACKEND == "azure":
        build_azure(client)
    else:
        index = LocalIndex.build(client)
        index.save()
        kinds = [d["kind"] for d in index.docs]
        print(f"Local index: {kinds.count('product')} products + {kinds.count('help')} help sections, "
              f"{index.vectors.shape[1]}-dimension vectors from {EMBED_MODEL}.")
    print(f"Done in {time.perf_counter() - start:.1f}s.")


if __name__ == "__main__":
    main()

def _upper_match(pattern, example):
    def check(cls, v):
        v = v.strip().upper()
        if not pattern.fullmatch(v):
            raise ValueError(f"must look like {example}")
        return v
    return classmethod(check)

@app.post("/api/chat")
def chat_endpoint(req: ChatRequest):
    """Stream the answer as server-sent events: one `data: {json}` line per event."""
    if req.customer_id is not None and req.customer_id not in backend.CUSTOMERS:
        raise HTTPException(status_code=400, detail="Unknown customer.")

    def events():
        for event in chat.stream(req.session_id, req.message, customer_id=req.customer_id):
            yield f"data: {json.dumps(event, ensure_ascii=False)}\n\n"
    return StreamingResponse(events(), media_type="text/event-stream",
                             headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})

                    k: int = Query(default=5, ge=1, le=20)):
    """New in milestone 5: semantic + keyword (hybrid) search over products or help-centre sections."""
    if search is None:
        raise HTTPException(status_code=503, detail="Search index not built. Run: python -m scripts.build_index")
    try:
        hits = search.search(q, k=k, kind=kind, family=family, max_annual_fee_usd=max_annual_fee_usd)
    except openai.APIError as e:
        log_event(level="error", event="search_api_error", error=type(e).__name__)
        raise HTTPException(status_code=503, detail="The embedding service is unavailable. Please try again shortly.")
    return {"query": q, "kind": kind, "results": [
        {k_: h.get(k_) for k_ in ("id", "title", "score", "source", "family", "annual_fee_usd")} | {"text": h["text"][:300]}
        for h in hits]}

    Tests with a fake embedding model

    Search code needs embeddings, but tests shouldn't call a model. FakeEmbeddings hashes words into a 256-dimension vector: texts that share words get similar vectors. It can't understand meaning, but it's deterministic, which is perfect for testing the plumbing (filters, ranking, save/load, fallbacks).

    class FakeEmbeddings:
    """Milestone 5: a deterministic 'embedding' (hashed bag of words), so texts that share
    words get similar vectors. Good enough to test the plumbing; real models capture meaning."""
    DIMS = 256

    def __init__(self):
        self.calls = 0

    def vector(self, text):
        v = [0.0] * self.DIMS
        for word in re.findall(r"[a-z0-9]+(?:-[a-z0-9]+)*", text.lower()):
            v[zlib.crc32(word.encode()) % self.DIMS] += 1.0
        v[0] += 0.01                                         # never an all-zero vector
        return v

    def create(self, model, input, **kwargs):
        self.calls += 1
        return NS(data=[NS(embedding=self.vector(t)) for t in input])
