Module 07 · Build milestone

Build · Milestone 7: Actions with the Learner's Approval

Version 0.7 doesn't just explain how to extend a lab or move a date; it prepares the change, shows the learner exactly what will happen, and does it when they press Approve. The model can only propose; code checks the rules, executes once and logs it. This is the Module 07 hands-on exercise: an agent that understands a request, selects and calls the right tools, processes the results and returns a final answer.

Propose, don't actApprove / DeclineIdempotent writesAgent guard rails≈ 75 min
Plan 10 min · Build 45 min · Verify 20 min
1

Plan

From telling learners what to do, to doing it with their consent.

TrainCrest· the brief for milestone 7

Lab extensions and date changes are a large share of support tickets, and each one waits about a day for a person who follows the same rules every time. The brief: let Traincrest Assist do these two actions itself, within policy. Constraints from the support lead: nothing changes without the learner's explicit approval; the learner sees the exact effect and fee before approving; a double click or a retry can never change a booking twice; refunds and exceptions still go to people; and every change is logged with who approved it.

# Lab access per enrolment, and the dates each course runs. Writes go through the two functions
# below, which re-check the rules, are idempotent (same key, same result, done once) and are logged.
LABS = {
    "TC-10293": {"lab_until": _day(4), "free_extension_used": False},
    "TC-31002": {"lab_until": _day(27), "free_extension_used": True},
}
SESSIONS = {"C104": [_day(21), _day(35)], "C106": [_day(10), _day(24), _day(38)],
            "C109": [_day(8), _day(22)], "C110": [_day(5), _day(17), _day(33)], "C112": [_day(12), _day(40)]}
ACTIONS_FILE = DATA / "actions.jsonl"
_DONE: dict[str, dict] = {}


class ActionError(Exception):
    """A rule says no. The message is safe to show to the learner."""


def upcoming_sessions(course_id: str) -> list[str]:
    return [d for d in SESSIONS.get(course_id, []) if d > date.today().isoformat()]


def _log(action: str, record: dict):
    ACTIONS_FILE.parent.mkdir(parents=True, exist_ok=True)
    with ACTIONS_FILE.open("a", encoding="utf-8") as f:
        f.write(json.dumps({"action": action, "at": date.today().isoformat(), **record}) + "\n")


def extend_lab(enrolment_id: str, key: str) -> dict:
    if key in _DONE:                                     # a retry of an action already done: same answer
        return _DONE[key]
    lab = LABS.get(enrolment_id)
    if lab is None:
        raise ActionError("This enrolment has no lab access to extend.")
    if lab["free_extension_used"]:
        raise ActionError("The free 7-day extension has already been used.")
    lab["lab_until"] = (date.fromisoformat(lab["lab_until"]) + timedelta(days=7)).isoformat()
    lab["free_extension_used"] = True
    _DONE[key] = result = {"enrolment_id": enrolment_id, "lab_until": lab["lab_until"], "fee_usd": 0}
    _log("extend_lab", {"key": key, **result})
    return result


def reschedule(enrolment_id: str, new_start: str, fee_usd: int, key: str) -> dict:
    if key in _DONE:
        return _DONE[key]
    e = ENROLMENTS[enrolment_id]
    if new_start not in upcoming_sessions(e["course_id"]) or new_start == e["start_date"]:
        raise ActionError(f"{new_start} is no longer an available date for this course.")
    old, e["start_date"] = e["start_date"], new_start
    _DONE[key] = result = {"enrolment_id": enrolment_id, "old_start": old, "new_start": new_start, "fee_usd": fee_usd}
    _log("reschedule", {"key": key, **result})
    return result
  2. Run

    The action store

    Proposals wait here. decide() enforces the approval rules from topic 7: right session and learner, one decision, expiry, re-check, execute once.

    """Actions that need the learner's approval (milestone 7).

The model can only PROPOSE a change (request_lab_extension, request_reschedule in tools.py).
A proposal is stored here with frozen arguments and a summary written by code. Nothing happens
until the learner presses Approve. Then code re-checks the rules and executes it once
(the action id is the idempotency key). The model never approves its own actions."""
import time
import uuid
from datetime import date

from app import backend, policy

EXPIRY_S = 15 * 60                     # a proposal older than this must be asked for again


def new_action_id() -> str:
    return "A-" + uuid.uuid4().hex[:6].upper()


class ActionStore:
    def __init__(self):
        self.pending: dict[str, dict] = {}         # action_id -> proposal (in memory; Module 08 persists it)

    def add(self, session_id: str, proposal: dict):
        self.pending[proposal["action_id"]] = {**proposal, "session_id": session_id, "created": time.time()}

    def for_session(self, session_id: str) -> list[dict]:
        return [p for p in self.pending.values() if p["session_id"] == session_id]

    def decide(self, session_id: str, action_id: str, learner_id: str | None, approve: bool) -> dict:
        """Returns {"status": done|declined|expired|failed|unknown, "message": text for the learner}."""
        p = self.pending.get(action_id)
        if p is None or p["session_id"] != session_id or p["learner_id"] != learner_id:
            return {"status": "unknown", "message": "That request isn't waiting for approval any more."}
        del self.pending[action_id]                # one decision per proposal, whatever happens next
        if not approve:
            return {"status": "declined", "message": f"OK, I haven't changed anything ({p['summary']})."}
        if time.time() - p["created"] > EXPIRY_S:
            return {"status": "expired", "message": "That request has expired. Please ask me again."}
        try:
            return {"status": "done", "message": EXECUTE[p["tool"]](p, action_id)}
        except backend.ActionError as e:
            return {"status": "failed", "message": f"I couldn't do that: {e}"}


def _extend(p, key):
    r = backend.extend_lab(p["args"]["enrolment_id"], key=key)
    return f"Done: your lab for {r['enrolment_id']} is now open until {r['lab_until']} (free)."


def _reschedule(p, key):
    e = backend.ENROLMENTS[p["args"]["enrolment_id"]]
    fee = policy.reschedule_fee(date.fromisoformat(e["start_date"]), date.today())   # re-check: time has passed
    if not fee["allowed"] or fee["fee_usd"] != p["fee_usd"]:
        raise backend.ActionError("the fee or rules changed since you were asked; please ask me again")
    r = backend.reschedule(p["args"]["enrolment_id"], p["args"]["new_start"], fee["fee_usd"], key=key)
    cost = f"${r['fee_usd']} fee" if r["fee_usd"] else "no fee"
    return f"Done: {r['enrolment_id']} now starts on {r['new_start']} (was {r['old_start']}; {cost})."


EXECUTE = {"request_lab_extension": _extend, "request_reschedule": _reschedule}

def _propose(context, tool, args, summary, **extra):
    """Record a proposal for the learner to approve. Nothing changes until they do."""
    proposal = {"action_id": actions.new_action_id(), "tool": tool, "args": args, "summary": summary,
                "learner_id": context["learner_id"], **extra}
    context.setdefault("proposals", []).append(proposal)
    return {"status": "awaiting_learner_approval", "action_id": proposal["action_id"], "summary": summary,
            "note": "Not done yet. Tell the learner to approve or decline with the buttons below your message."}

    return _propose(context, "request_reschedule",
                    {"enrolment_id": args.enrolment_id.strip().upper(), "new_start": args.new_start_date},

                    fee_usd=fee["fee_usd"])

def looks_like_tool_call(text: str) -> bool:
    """Small models sometimes write a tool call as text ('{"name": "quote_refund", ...}') instead of
    calling it. Never show that to a learner (milestone 7)."""
    t = text.strip().strip("`").removeprefix("json").strip()
    return t.startswith("{") and '"name"' in t and ('"arguments"' in t or '"parameters"' in t)


def learner_context(learner_id: str) -> str:
    """Who is signed in and their own enrolments, from the app, not the model (milestone 7). Saves a tool
    call on almost every request, and contains nothing the learner isn't allowed to see."""
    rows = backend.enrolments_of(learner_id)
    lines = [f"- {r['enrolment_id']}: {r['course_title']} (course {r['course_id']}), starts {r['start_date']}, {r['status']}"
             for r in rows]
    return (f"## Signed-in learner\n{backend.LEARNERS[learner_id]['name']} ({learner_id}). Their enrolments:\n"
            + "\n".join(lines))


        def decide(self, session_id: str, action_id: str, learner_id: str | None, approve: bool) -> dict:
