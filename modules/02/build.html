Build · Milestone 2: A Streaming Chat Service

Version 0.2 turns the CLI into a real service: a FastAPI backend that streams answers to a browser chat, remembers the conversation, survives failures with friendly messages, retries what's transient, and logs latency, tokens and cost for every turn. You write the backend; your AI assistant builds the UI to your contract.

    """The only place in Traincrest Assist that knows how to reach the model."""
import os

from openai import OpenAI

from app.config import PROVIDER


def get_client() -> OpenAI:
    if PROVIDER == "azure":
        client = OpenAI(base_url=os.environ["AZURE_OPENAI_BASE_URL"], api_key=os.environ["AZURE_OPENAI_API_KEY"])
    elif PROVIDER == "openai":
        client = OpenAI()
    elif PROVIDER == "ollama":
        client = OpenAI(base_url="http://localhost:11434/v1", api_key="ollama")
    else:
        raise ValueError(f"Unknown LLM_PROVIDER: {PROVIDER!r}")
    # New in milestone 2: the SDK retries 408/409/429/5xx with backoff; give it 3 tries and a timeout.
    return client.with_options(max_retries=3, timeout=60)

    """Structured logging and cost: one JSON line per event, ready for any log platform."""
import json
import logging
import time

from app.config import LOG_FILE, PRICE_IN_PER_M, PRICE_OUT_PER_M

_logger = logging.getLogger("traincrest_assist")
if not _logger.handlers:
    _handler = logging.FileHandler(LOG_FILE, encoding="utf-8")
    _handler.setFormatter(logging.Formatter("%(message)s"))
    _logger.addHandler(_handler)
    _logger.setLevel(logging.INFO)
    _logger.propagate = False


def log_event(**fields):
    """Write one JSON object per line. Never log API keys or full learner messages."""
    _logger.info(json.dumps({"ts": time.strftime("%Y-%m-%dT%H:%M:%S"), **fields}, ensure_ascii=False))


def cost_usd(usage):
    """Dollar cost of one call from its token usage."""
    if usage is None:
        return None
    return (usage.prompt_tokens * PRICE_IN_PER_M + usage.completion_tokens * PRICE_OUT_PER_M) / 1_000_000

    The heart: ChatService

    Everything from this module meets here: history trimming (Drill 02.6), streaming with the empty-chunk guard (Drill 02.8), the error mapping (Drill 02.11), and usage and cost (Drill 02.14). Read it top to bottom before you run it.

    """Chat service: conversation memory, streaming, error handling and usage logging.
Kept separate from the web layer so it can be used from the CLI, the API and tests."""
import time
import uuid

import openai

from app.config import MAX_TURNS_KEPT, MODEL
from app.llm import get_client
from app.logging_utils import cost_usd, log_event
from app.prompts import SYSTEM_PROMPT

FRIENDLY = {
    "rate_limited": "Traincrest Assist is very busy right now. Please try again in a moment.",
    "unreachable": "I couldn't reach the AI service. Please try again.",
    "bad_request": "I couldn't process that message. Could you rephrase it?",
    "config": "Traincrest Assist is temporarily unavailable. Our team has been notified.",
    "server": "Something went wrong on our side. Please try again shortly.",
}


class ChatService:
    def __init__(self, client=None):
        self.client = client or get_client()
        # session_id -> list of messages. In memory for now; Module 08 moves this to a database.
        self.sessions: dict[str, list[dict]] = {}

    def reset(self, session_id: str) -> None:
        self.sessions.pop(session_id, None)

    def stream(self, session_id: str, user_message: str):
        """Yield events for one turn: {"type": "delta", "text": ...} while the answer streams,
        then exactly one {"type": "done", ...} or {"type": "error", "message": ...}."""
        history = self.sessions.setdefault(session_id, [])
        history[:] = history[-(MAX_TURNS_KEPT * 2):]            # keep whole user/assistant pairs
        history.append({"role": "user", "content": user_message})
        turn_id, start = uuid.uuid4().hex[:8], time.perf_counter()
        ttft, parts, usage, finish = None, [], None, None
        error, detail = None, ""

        try:
            stream = self.client.chat.completions.create(
                model=MODEL,
                messages=[{"role": "system", "content": SYSTEM_PROMPT}] + history,
                stream=True,
                stream_options={"include_usage": True},
            )
            for chunk in stream:
                if chunk.usage:                                  # final chunk carries token usage
                    usage = chunk.usage
                if not chunk.choices:                            # filter-only or usage-only chunks
                    continue
                choice = chunk.choices[0]
                finish = choice.finish_reason or finish
                if choice.delta.content:
                    ttft = ttft or time.perf_counter() - start
                    parts.append(choice.delta.content)
                    yield {"type": "delta", "text": choice.delta.content}
        except openai.RateLimitError as e:
            error, detail = "rate_limited", str(e)
        except openai.APIConnectionError as e:                   # includes timeouts
            error, detail = "unreachable", str(e)
        except openai.BadRequestError as e:
            error, detail = "bad_request", str(e)
        except (openai.AuthenticationError, openai.PermissionDeniedError, openai.NotFoundError) as e:
            error, detail = "config", str(e)
        except openai.APIStatusError as e:
            error, detail = "server", str(e)

        if error:
            history.pop()                                        # forget the unanswered question
            log_event(turn=turn_id, session=session_id[:8], level="error", error=error, detail=detail[:200])
            yield {"type": "error", "message": FRIENDLY[error]}
            return

        reply = "".join(parts)
        history.append({"role": "assistant", "content": reply})
        cost = cost_usd(usage)
        event = {
            "type": "done",
            "finish_reason": finish,
            "ttft_s": round(ttft or 0, 2),
            "total_s": round(time.perf_counter() - start, 2),
            "tokens_in": getattr(usage, "prompt_tokens", None),
            "tokens_out": getattr(usage, "completion_tokens", None),
            "cost_usd": round(cost, 6) if cost is not None else None,
        }
        log_event(turn=turn_id, session=session_id[:8], level="info", model=MODEL,
                  **{k: v for k, v in event.items() if k != "type"})
        yield event
  4. Run

    The web API

    """Traincrest Assist API, milestone 2.

Run from the project folder:  uvicorn app.main:app --reload
Then open http://127.0.0.1:8000 (chat UI) or http://127.0.0.1:8000/docs (API docs).
"""
import json
from pathlib import Path

from fastapi import FastAPI
from fastapi.responses import FileResponse, StreamingResponse
from pydantic import BaseModel, Field

from app.chat import ChatService

app = FastAPI(title="Traincrest Assist", version="0.2")
chat = ChatService()
WEB = Path(__file__).resolve().parent.parent / "web"


class ChatRequest(BaseModel):
    session_id: str = Field(min_length=8, max_length=64)
    message: str = Field(min_length=1, max_length=4000)


class ResetRequest(BaseModel):
    session_id: str = Field(min_length=8, max_length=64)


@app.get("/")
def home():
    return FileResponse(WEB / "index.html")


@app.get("/api/health")
def health():
    return {"status": "ok"}


@app.post("/api/chat")
def chat_endpoint(req: ChatRequest):
    """Stream the answer as server-sent events: one `data: {json}` line per event."""
    def events():
        for event in chat.stream(req.session_id, req.message):
            yield f"data: {json.dumps(event, ensure_ascii=False)}\n\n"
    return StreamingResponse(events(), media_type="text/event-stream",
                             headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})


@app.post("/api/reset")
def reset(req: ResetRequest):
    chat.reset(req.session_id)
    return {"status": "reset"}
  5. Ask AI

    Have your assistant build the chat page to your contract

    Create web/index.html: a single-file chat page (HTML, CSS and vanilla JavaScript, no frameworks,
no external files) for "Traincrest Assist". Contract with the backend in #app/main.py:
- On load, create sessionId = crypto.randomUUID().
- On submit, POST /api/chat with JSON {session_id, message}. The response is server-sent events:
  lines "data: {json}" separated by a blank line. Read it with fetch + response.body.getReader()
  + TextDecoder, buffering until "\n\n".
- Event types: {"type":"delta","text"} append text to the current assistant bubble;
  {"type":"replace","text"} replace the bubble's text; {"type":"error","message"} show the message
  styled as an error; {"type":"done","ttft_s","total_s","tokens_in","tokens_out"} show a small
  grey line under the bubble with those numbers.
- If fetch fails or the status isn't OK, show a friendly error in the bubble.
- A "New chat" button POSTs /api/reset {session_id}, makes a new sessionId and clears the log.
- Disable Send while a reply is streaming. Use textContent, never innerHTML, for message text.
Dark, clean design; user bubbles on the right, assistant on the left.
    Why this prompt works: it's a contract, not a wish. It names the exact endpoint, payload, wire format and every event type (including replace, which Milestone 3 needs), says how to parse a streamed body, and includes a security rule: textContent instead of innerHTML, so model output can never inject HTML or scripts into the page.

<script>
  let sessionId = crypto.randomUUID();
  const log = document.getElementById("log");
  const form = document.getElementById("form");
  const input = document.getElementById("input");
  const send = document.getElementById("send");

  function add(cls, text) {
    const el = document.createElement("div");
    el.className = cls;
    el.textContent = text;
    log.appendChild(el);
    log.scrollTop = log.scrollHeight;
    return el;
  }

  function handle(event, bubble) {
    if (event.type === "delta") bubble.textContent += event.text;
    else if (event.type === "replace") bubble.textContent = event.text;
    else if (event.type === "error") { bubble.textContent = event.message; bubble.classList.add("error"); }
    else if (event.type === "done") {
      add("meta", `first word ${event.ttft_s}s · total ${event.total_s}s · ${event.tokens_in ?? "?"} in / ${event.tokens_out ?? "?"} out`);
    }
    log.scrollTop = log.scrollHeight;
  }

  form.addEventListener("submit", async (e) => {
    e.preventDefault();
    const text = input.value.trim();
    if (!text) return;
    input.value = "";
    send.disabled = true;
    add("msg you", text);
    const bubble = add("msg bot", "");
    try {
      const res = await fetch("/api/chat", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ session_id: sessionId, message: text }),
      });
      if (!res.ok) throw new Error("HTTP " + res.status);
      const reader = res.body.getReader();
      const decoder = new TextDecoder();
      let buffer = "";
      while (true) {
        const { value, done } = await reader.read();
        if (done) break;
        buffer += decoder.decode(value, { stream: true });
        let cut;
        while ((cut = buffer.indexOf("\n\n")) >= 0) {        // SSE events end with a blank line
          const raw = buffer.slice(0, cut);
          buffer = buffer.slice(cut + 2);
          if (raw.startsWith("data: ")) handle(JSON.parse(raw.slice(6)), bubble);
        }
      }
    } catch (err) {
      handle({ type: "error", message: "Couldn't reach Traincrest Assist (" + err.message + ")." }, bubble);
    } finally {
      send.disabled = false;
      input.focus();
    }
  });

  document.getElementById("reset").addEventListener("click", async () => {
    await fetch("/api/reset", { method: "POST", headers: { "Content-Type": "application/json" },
                                body: JSON.stringify({ session_id: sessionId }) });
    sessionId = crypto.randomUUID();
    log.innerHTML = "";
  });

  6. Run

    Upgrade the CLI to share the same service

    """Traincrest Assist in the terminal, milestone 2: same ChatService as the web app, streamed.

Run from the project folder:  python cli.py
"""
import uuid

from app.chat import ChatService
from app.config import MODEL, PROVIDER


def main():
    chat, session = ChatService(), uuid.uuid4().hex
    print(f"Traincrest Assist v0.2 [{PROVIDER} · {MODEL}]. Type 'exit' to quit, '/reset' for a new chat.\n")
    while True:
        try:
            question = input("You: ").strip()
        except (EOFError, KeyboardInterrupt):
            break
        if question.lower() in {"exit", "quit"}:
            break
        if question == "/reset":
            chat.reset(session)
            print("(new conversation)\n")
            continue
        if not question:
            continue
        print("Assist: ", end="", flush=True)
        for event in chat.stream(session, question):
            if event["type"] == "delta":
                print(event["text"], end="", flush=True)
            elif event["type"] == "error":
                print(event["message"])
            elif event["type"] == "done":
                print(f"\n  ({event['ttft_s']}s to first word · {event['total_s']}s · "
                      f"{event['tokens_in']} in / {event['tokens_out']} out · ${event['cost_usd']})")
        print()


if __name__ == "__main__":
    main()
